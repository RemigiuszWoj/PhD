\section{Introduction}
\begin{frame}{Introduction}
    \begin{itemize}
        \item \textbf{Permutation Flow Shop Problem (PFSP):} NP-hard for $m \geq 3$. Metaheuristics rely on a chosen \emph{neighborhood} to explore the solution space.
        \item Single-neighborhood local search has systematic blind spots:
            \begin{itemize}
                \item \textbf{Adjacent swap:} cheap, but only short-range moves.
                \item \textbf{Dynasearch:} expensive per iteration.
                \item \textbf{Motzkin arcs:} cover specific structural patterns only.
            \end{itemize}
        \item \textbf{Idea:} combine multiple elementary neighborhoods into a hybrid scheme that adapts to the current state of the search.
        \item This work: five hybridization strategies, two metaheuristic frameworks (TS, SA), four elementary neighborhoods --- on Taillard benchmarks.
    \end{itemize}
\end{frame}

% ===== SLIDE 4 =====
\section{Elementary Neighborhoods}
\begin{frame}{Problem and Elementary Neighborhoods}
    \textbf{PFSP makespan:} minimize $C_{\max}(\pi) = C_{m{-}1,n{-}1}(\pi)$ over $\pi \in S_n$.

    \vspace{0.3cm}
    Four elementary neighborhoods:
    \begin{center}\small
    \begin{tabular}{lll}
        \toprule
        Symbol & Move & Size / cost \\
        \midrule
        $N_{\mathrm{adj}}$  & swap $\pi_i \leftrightarrow \pi_{i+1}$               & $n{-}1$,\quad $O(1)$ per $\Delta$ \\
        $N_{\mathrm{fib}}$  & reinsert at $i + F_k$ (Fibonacci jumps)              & $O(n\log n)$, selective long-range \\
        $N_{\mathrm{dyn}}$  & DP-optimal composite swap                            & $O(n^2)$ per application \\
        $N_{\mathrm{motz}}$ & non-crossing arc moves                               & $M_n \sim 3^n/n^{3/2}$, filtered \\
        \bottomrule
    \end{tabular}
    \end{center}

    \vspace{0.3cm}
    All neighborhoods accelerated by the Smutnicki block-property and the NPI filter.
\end{frame}

% ===== SLIDE 5 =====
\section{Hybridization Strategies}
\begin{frame}{Five Hybridization Strategies}
    Let $\mathcal{N} = \{N_1, \dots, N_K\}$. At iteration $t$ select one $N_{s(t)}$:

    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{FC --- Fixed Cycle:} $s(t) = (t \bmod K) + 1$. Baseline.
        \item \textbf{PW --- Probability-Weighted:} $s(t) \sim \mathrm{Categorical}(\mathbf{w})$ with tuned weights.
        \item \textbf{PA --- Performance-Adaptive:} $\varepsilon$-greedy on EMA of past improvements per neighborhood.
        \item \textbf{BK --- Best-of-$K$:} evaluate all $K$ neighborhoods, take the best move. Best quality, $K\times$ cost.
        \item \textbf{DP --- Dynamic Programming:} finite-horizon MDP over $(\pi, t)$; backward induction with horizon $H = 5$. Only feasible for $n \leq 50$.
    \end{itemize}
\end{frame}

% ===== SLIDE 6 =====
\begin{frame}{Performance-Adaptive Strategy --- in Detail}
    EMA of improvement per neighborhood:
    \[
        \bar{\delta}_k^{(t+1)} \;=\; \gamma\,\bar{\delta}_k^{(t)} \;+\; (1-\gamma)\,\delta_k^{(t)},\qquad \gamma = 0.9
    \]
    where $\delta_k^{(t)}$ is the actual $\Delta C_{\max}$ gained by $N_k$ at step $t$ (0 if not selected).

    \vspace{0.2cm}
    Selection rule ($\varepsilon$-greedy):
    \[
        s(t) \;=\;
        \begin{cases}
          \arg\max_k \bar{\delta}_k^{(t)} & \text{with probability } 1-\varepsilon, \\
          \text{uniform random} \in \mathcal{N} & \text{otherwise}.
        \end{cases}
    \]

    \vspace{0.2cm}
    \textbf{Why it works:} matches \textbf{Best-of-$K$} quality at $1/K$ of the per-iteration cost --- it learns the best neighborhood for the current basin instead of trying all every step.
\end{frame}

% ===== SLIDE 7 =====
\section{Algorithm}
\begin{frame}{Hybrid Tabu Search (HTS)}
    \begin{algorithm}[H]
    \SetAlgoLined
    \SetKwInOut{Input}{input}\SetKwInOut{Output}{output}
    \Input{initial $\pi_0$, time budget $T_{\max}$}
    \Output{best permutation $\pi^*$}
    $\pi \gets \pi_0$;\quad $\pi^* \gets \pi$;\quad $\mathcal{T} \gets \emptyset$;\quad $\mathcal{M} \gets \mathrm{MushroomList}(k{=}10)$\;
    \While{$\text{time} < T_{\max}$}{
        select $N_{s(t)}$ by hybridization strategy\;
        $(\pi', m) \gets$ best non-tabu move in $N_{s(t)}(\pi)$ (or aspiration if it beats $\pi^*$)\;
        \eIf{$m = \emptyset$}{
            $\pi \gets \mathcal{M}.\mathrm{perturb}()$ \quad\textit{// double-bridge from elite pool}\;
        }{
            $\pi \gets \pi'$;\quad $\mathcal{T} \gets \mathcal{T} \cup \{m\}$\;
        }
        update $\pi^*$, $\mathcal{M}$, and EMAs $\bar{\delta}_k$\;
    }
    \Return $\pi^*$\;
    \end{algorithm}
\end{frame}

\section{Experiments}
\begin{frame}{Experimental Setup}
    \begin{itemize}
        \item \textbf{Benchmarks:} Taillard instances, $n \in \{20, 50, 100, 200\}$, $m \in \{5, 10, 20\}$, 10 instances per class.
        \item \textbf{Time budget:} $T_{\max} = 60\,000$~ms.
        \item \textbf{Metaheuristics:} Tabu Search ($\tau = n/5$, aspiration, MushroomList $k{=}10$) and Simulated Annealing (exp.\ cooling, $\alpha = 0.995$, adaptive reheat).
        \item \textbf{Quantum variants:} D-Wave Advantage 4.1, $5{,}000$ reads, windowed decomposition for $n > 19$ (window 19, overlap 9).
        \item \textbf{Metric:} mean relative percentage deviation (RPD) to best-known $C_{\max}^{*}$:
        \[
            \mathrm{RPD} \;=\; \frac{C_{\max} - C_{\max}^{*}}{C_{\max}^{*}} \times 100\%.
        \]
    \end{itemize}
\end{frame}

% ===== SLIDE 10 =====
\section{Conclusion}
\begin{frame}{Conclusion and Future Work}
    \textbf{Summary:}
    \begin{itemize}
        \item Hybrid neighborhoods consistently beat single-neighborhood baselines in both TS and SA.
        \item \textbf{Performance-Adaptive (PA)} is the sweet spot: best-of-$K$ quality at $1/K$ of the per-iteration cost.
        \item Combining Motzkin and Fibonacci alongside Adjacent and Dynasearch matters most for large instances ($n = 200$).
        \item Quantum-assisted neighborhoods give an extra $0.08$--$0.15$\,pp at $n = 200$.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Future work:}
    \begin{itemize}
        \item Multi-armed bandit formulations for neighborhood selection (UCB, Thompson sampling).
        \item Quantum-native hybrid strategies exploiting D-Wave's native graph structure.
        \item Joint hybridization across TS and SA (portfolio approach).
    \end{itemize}
\end{frame}
