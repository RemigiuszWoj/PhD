\begin{block}{1.\ Problem}
\textbf{Permutation Flow Shop Problem (PFSP).} $n$ jobs processed in the
same order on $m$ machines; minimize the makespan
\[
  C_{\max}(\pi) \;=\; C_{m,n}(\pi).
\]
NP-hard for $m \geq 3$ — metaheuristics are the standard tool.

\vspace{0.4em}
\textbf{Question.} Inside Tabu Search (TS) and Simulated Annealing (SA),
which \emph{neighborhood structure} performs best under a fixed wall-clock
time budget?
\end{block}

\begin{block}{2.\ Neighborhood structures}
\begin{center}

\vspace{0.4em}
\textbf{Composite Disjoint-Adjacent (\emph{Fibonacci}).}
A move is a subset $S \subseteq \{1,\dots,n{-}1\}$ of adjacent-swap
positions with the \textbf{non-overlap} constraint
\[
  i \in S \;\Longrightarrow\; i+1 \notin S.
\]
All selected swaps $\pi_i \leftrightarrow \pi_{i+1}$ are applied
\emph{simultaneously}.

\vspace{0.5em}
\begin{center}
\begin{tikzpicture}[scale=1.4]
  \foreach \x in {0,1,2,3,4,5,6,7} \fill (\x,0) circle (4pt) node[below=6pt] {\large\x};
  \draw[very thick,pwrone] (0,0) to[bend left=80] (1,0);
  \draw[very thick,pwrone] (2,0) to[bend left=80] (3,0);
  \draw[very thick,pwrone] (5,0) to[bend left=80] (6,0);
  \node[above=4pt,pwrone] at (0.5,0.6) {\small swap};
  \node[above=4pt,pwrone] at (2.5,0.6) {\small swap};
  \node[above=4pt,pwrone] at (5.5,0.6) {\small swap};
\end{tikzpicture}
\end{center}

\vspace{0.3em}
\textbf{Counting.} The number of valid subsets in a permutation of length
$n$ equals the Fibonacci number $F_{n+1}$ — hence the nickname.
\end{block}

\begin{block}{3.\ Algorithm \& cost}
Per iteration:
\begin{enumerate}
  \item Recompute $C_{\max}(\pi^{(i)})$ for each $i = 1,\dots,n{-}1$ — each $O(mn)$.
  \item One-pass DP picks the optimal non-overlapping subset:\\[0.3em]
        \quad $\mathrm{dp}[i] = \min\!\bigl(\mathrm{dp}[i+1],\ \Delta_i + \mathrm{dp}[i+2]\bigr)$.
  \item Apply selected swaps; fall back to the least positive $\Delta_i$ if DP picks none.
\end{enumerate}

\vspace{0.6em}
\begin{center}
\begin{tabular}{lll}
  \toprule
  \textbf{Neighborhood} & \textbf{Candidates} & \textbf{Per iter.} \\
  \midrule
  Adjacent              & $n-1$ swaps        & $O(mn)$ \\
  \textbf{\emph{Fibonacci}} & $n-1$ swaps + DP   & $O(mn^2)$ \\
  Dynasearch (rec.)     & $O(n^2)$ segments  & $O(mn^3)$ \\
  \bottomrule
\end{tabular}
\end{center}

\vspace{0.5em}
\emph{Fibonacci} explores an \textbf{exponentially-sized} neighborhood
($F_{n+1}$ valid subsets) at \textbf{quadratic} per-iteration cost — the
sweet spot between cheap Adjacent and expensive Dynasearch.
\end{block}

\begin{block}{4.\ Metaheuristics}
\textbf{Tabu Search (TS).} Tabu list keyed by primitive moves or composite
signatures; tenure $\tau_{\text{tabu}} = 10$; aspiration on global-best
improvement.

\vspace{0.6em}
\textbf{Simulated Annealing (SA).} Multiplicative cooling $T \leftarrow \alpha T$
with floor; \textbf{time-based reheating}: if no improvement for
$\tau_{\mathrm{stagn}} = 500$\,ms and $T < T_0$, set
$T \leftarrow \min(T_0, r \cdot T)$, $r = 1.5$.

\vspace{0.6em}
Same neighborhoods plugged into both frameworks — any performance gap is
attributable to the neighborhood, not the framework.
\end{block}

\begin{block}{5.\ Experimental setup}
\begin{itemize}
  \item Taillard instances, $n \in \{20, 50, 100, 200, 500\}$,
        $m \in \{5, 10, 20\}$ — 12 instance classes.
  \item Time limits: $tl \in \{100, 500, 1000, 2000, 5000, 10000\}$\,ms.
  \item SA: $T_0 = 1000$, $T_{\mathrm{final}} = 1$, $\alpha = 0.95$,
        reheat $r = 1.5$, stagnation 500\,ms.
  \item Platform: MacBook Pro M4 Pro, 24\,GB RAM, Python 3.11.
  \item Metric: PRD to best-known $C_{\max}^{*}$:\\
        $\mathrm{PRD} = \tfrac{C_{\max} - C_{\max}^{*}}{C_{\max}^{*}} \cdot 100\%$.
\end{itemize}
\end{block}

\begin{block}{8.\ Discussion}
\begin{itemize}
  \item \textbf{Early-budget gain.} Composite move bundles several
        non-overlapping improving swaps into one iteration.
  \item \textbf{Plateau risk.} Once all $\Delta_i \geq 0$, the composite
        degenerates to a single swap; SA reheating mitigates this.
  \item \textbf{Dynasearch cost.} A single iteration at $n = 500$ can
        exceed $tl$ by $\sim 200\times$.
  \item \textbf{Practical takeaway.} \emph{Fibonacci} is the right default
        for medium-to-large $n$ under tight time limits; Adjacent for very
        small $n$; Dynasearch only when ample budget is available.
\end{itemize}
\end{block}

\begin{block}{9.\ Conclusion \& future work}
\textbf{Summary.} Three neighborhoods compared inside TS and SA on
Taillard. \emph{Fibonacci} (Composite Disjoint-Adjacent) dominates from
$n = 200$ onward; Dynasearch reaches deeper optima but is impractical for
large $n$ within short time budgets.

\vspace{0.5em}
\textbf{Future work.}
\begin{itemize}
  \item Incremental delta evaluation for the composite move ($O(mn^2) \to O(mn)$).
  \item Adaptive switching between neighborhoods at runtime.
  \item Comparison with state-of-the-art FSP solvers (iterated LS, VND).
  \item Quantum-assisted Fibonacci selection on D-Wave (windowed QUBO,
        $n$ up to 500).
\end{itemize}
\end{block}
