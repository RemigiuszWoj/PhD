\section{Introduction}
\begin{frame}{Introduction}
    \textbf{Permutation Flow Shop Problem (PFSP)} --- $n$ jobs, $m$ machines, same order, minimize $C_{\max}$. NP-hard for $m \geq 3$.

\section{Neighborhood Structures}
\begin{frame}{Composite Disjoint-Adjacent (\emph{Fibonacci})}
    Subset $S \subseteq \{1,\dots,n{-}1\}$ of \textbf{non-overlapping} adjacent-swap positions; all selected swaps applied simultaneously. Number of valid subsets $= F_{n+1}$ --- hence the nickname.

    \vspace{0.3cm}
    \begin{center}
    \begin{tikzpicture}[scale=1.1]
        \foreach \x in {0,1,2,3,4,5,6,7} \fill (\x,0) circle (2.5pt) node[below=4pt] {\small\x};
        \draw[thick,pwrone] (0,0) to[bend left=80] (1,0);
        \draw[thick,pwrone] (2,0) to[bend left=80] (3,0);
        \draw[thick,pwrone] (5,0) to[bend left=80] (6,0);
        \node[above=3pt,pwrone] at (0.5,0.55) {\scriptsize swap};
        \node[above=3pt,pwrone] at (2.5,0.55) {\scriptsize swap};
        \node[above=3pt,pwrone] at (5.5,0.55) {\scriptsize swap};
    \end{tikzpicture}
    \end{center}

    \vspace{0.2cm}
    Optimal subset (minimal $\sum_i \Delta_i$) chosen by DP in $O(n)$ given the precomputed $n{-}1$ swap deltas.
\end{frame}

% ===== SLIDE 5 =====
\begin{frame}{Cost Comparison}
    \begin{center}
    \begin{tabular}{lll}
        \toprule
        Neighborhood & Candidates & Total per iteration \\
        \midrule
        Adjacent             & $n-1$ swaps        & $O(mn)$ \\
        \textbf{\emph{Fibonacci}} & $n-1$ swaps + DP   & $O(mn^2)$ \\
        Dynasearch (rec.)    & $O(n^2)$ segments  & $O(mn^3)$ \\
        \bottomrule
    \end{tabular}
    \end{center}

    \vspace{0.5cm}
    \emph{Fibonacci} explores an \textbf{exponentially-sized} neighborhood ($F_{n+1}$ valid subsets) at \textbf{quadratic} cost --- the sweet spot between cheap Adjacent and expensive Dynasearch.
\end{frame}

\section{Metaheuristics}
\begin{frame}{Tabu Search \& Simulated Annealing}
    \begin{columns}
        \begin{column}{.5\textwidth}
            \textbf{Tabu Search}
            \begin{itemize}
                \item Tabu list keyed by move or composite signature.
                \item Tenure $\tau_{\text{tabu}} = 10$.
                \item Aspiration on global best improvement.
            \end{itemize}
        \end{column}
        \begin{column}{.5\textwidth}
            \textbf{Simulated Annealing}
            \begin{itemize}
                \item Multiplicative cooling $T \leftarrow \alpha T$.
                \item \textbf{Time-based reheating} on stagnation ($\tau_{\mathrm{stagn}}=500$~ms).
                \item Standard Metropolis acceptance.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.5cm}
    \centering Same neighborhoods $\Rightarrow$ performance differences attributable to neighborhood design, not framework.
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion and Future Work}
    \textbf{Summary:}
    \begin{itemize}
        \item Three neighborhoods compared inside TS and SA on Taillard.
        \item \emph{Fibonacci} (Composite Disjoint-Adjacent) dominates from $n = 200$ onward.
        \item Dynasearch reaches deeper optima but is impractical for $n \geq 200$.
        \item SA's time-based reheating helps escape composite-move plateaus.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Future work:}
    \begin{itemize}
        \item Incremental delta evaluation for the composite move.
        \item Adaptive switching between neighborhoods at runtime.
        \item Comparison with state-of-the-art FSP solvers (iterated LS, VND).
    \end{itemize}
\end{frame}
