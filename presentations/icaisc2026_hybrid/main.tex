\section{Introduction}
\begin{frame}{Introduction}
    \begin{itemize}
        \item \textbf{Permutation Flow Shop Problem (PFSP):} NP-hard for $m \geq 3$. Neighborhood choice drives metaheuristic quality.
        \item Classical neighborhoods trade off coverage and cost:
            \begin{itemize}
                \item \textbf{Adjacent swap:} $n{-}1$ moves, only one swap per iteration.
                \item \textbf{Dynasearch:} $O(n^2)$ candidate pairs, expensive --- intractable beyond $n \sim 200$.
            \end{itemize}
        \item \textbf{Fibonacci neighborhood:} all sets of \emph{non-overlapping adjacent swaps} applied simultaneously --- $F_{n+1}$ valid subsets, but the optimal subset is selected by DP in $O(n)$.
        \item This talk: define the Fibonacci neighborhood, study its scaling behavior on Taillard benchmarks, and discuss its role within local search frameworks (ILS, SA).
    \end{itemize}
\end{frame}

% ===== SLIDE 4 =====
\section{Fibonacci Neighborhood}
\begin{frame}{The Fibonacci Neighborhood --- Definition}
    \textbf{Definition.} Given $\pi = (\pi_0, \dots, \pi_{n-1})$, a Fibonacci move is a subset $S \subseteq \{1, \dots, n{-}1\}$ of adjacent-swap positions such that
    \[
        i \in S \;\Longrightarrow\; i+1 \notin S \qquad \text{(no two consecutive swaps)}.
    \]
    All selected swaps $\pi_i \leftrightarrow \pi_{i+1}$ are applied \emph{simultaneously}.

    \vspace{0.2cm}
    \textbf{Example} (admissible: swap at 0, 2, 5; positions 1, 3, 4, 6 idle):
    \begin{center}
    \begin{tikzpicture}[scale=0.95]
        \foreach \x in {0,1,2,3,4,5,6,7} \fill (\x,0) circle (2pt) node[below=4pt] {\small\x};
        \draw[thick,pwrone] (0,0) to[bend left=80] (1,0);
        \draw[thick,pwrone] (2,0) to[bend left=80] (3,0);
        \draw[thick,pwrone] (5,0) to[bend left=80] (6,0);
    \end{tikzpicture}
    \end{center}

    \textbf{Counting.} The number of valid subsets in a permutation of length $n$ equals the Fibonacci number $F_{n+1}$ --- hence the name.
\end{frame}

% ===== SLIDE 5 =====
\begin{frame}{Properties and Computation}
    \begin{itemize}
        \item \textbf{Exponential size, linear cost.} $|N_{\mathrm{fib}}(\pi)| = F_{n+1} \sim \varphi^n / \sqrt{5}$, yet the optimal subset is found in $O(n)$.
        \item \textbf{Algorithm:} compute the $n{-}1$ adjacent-swap deltas using Head+Tail in $O(mn)$, then a one-pass DP over the deltas:
        \[
            \mathrm{dp}[i] \;=\; \min\!\bigl(\,\mathrm{dp}[i+1],\ \delta_i + \mathrm{dp}[i+2]\,\bigr)
        \]
        decides at each position whether to take the swap or skip it.
        \item \textbf{Acceleration:} the Smutnicki block-property and the NPI filter prune dominated swaps before the DP --- typical reduction $30$--$60\%$.
        \item \textbf{Compared with Dynasearch.} Same compound-move idea, but Fibonacci only considers adjacent pairs $(i,i{+}1)$ while Dynasearch considers arbitrary $(i,j)$ --- $n{-}1$ candidates vs.\ $O(n^2)$.
    \end{itemize}
\end{frame}

% ===== SLIDE 6 =====
\section{Quantum Formulation}
\begin{frame}{QUBO Formulation for D-Wave}
    Binary variable $x_i \in \{0,1\}$ for each adjacent-swap position $i \in \{1,\ldots,n{-}1\}$: $x_i = 1$ iff swap $\pi_i \leftrightarrow \pi_{i+1}$ is applied.

    \vspace{0.2cm}
    \textbf{QUBO form} --- tridiagonal coupling:
    \[
        Q_{\mathrm{fib}}(\mathbf{x}) \;=\; \sum_{i=1}^{n-1} \delta_i\, x_i
        \;+\; P \sum_{i=1}^{n-2} x_i\, x_{i+1},
    \]
    where $\delta_i$ is the precomputed adjacent-swap cost and $P \gg \max |\delta_i|$ penalizes overlapping swaps (encoding the non-overlapping constraint).

    \vspace{0.2cm}
    \textbf{On D-Wave Advantage:}
    \begin{itemize}
        \item Bandwidth 2 $\Rightarrow$ linear-chain embedding on Pegasus/Zephyr, no chain breaks.
        \item Logical qubits $=n{-}1$; feasible up to $n \leq 500$ on current hardware.
    \end{itemize}
\end{frame}

\section{Experiments}
\begin{frame}{Experimental Setup}
    \begin{itemize}
        \item \textbf{Benchmarks:} Taillard instances, $n \in \{20, 50, 100, 200, 500\}$, $m \in \{5, 10, 20\}$, 10 instances each, 5 seeds.
        \item \textbf{Time limits:} $t_{\max} \in \{0.1, 0.5, 1, 2, 5, 10\}$~s; metric: mean RPD to lower bound.
        \item \textbf{Algorithms:} Iterated Local Search (tabu $\tau{=}10$, MushroomList $k{=}10$); Simulated Annealing ($T_0{=}1000$, $\alpha{=}0.99$, reheat-kick on stagnation).
        \item \textbf{Quantum:} D-Wave Advantage 4.1, $5{,}000$ reads, windowed decomposition for $n > 19$.
        \item \textbf{Compared neighborhoods:} Adjacent, \textbf{Fibonacci}, Dynasearch.
    \end{itemize}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion and Future Work}
    \textbf{Summary:}
    \begin{itemize}
        \item Fibonacci selects an optimal set of non-overlapping adjacent swaps applied simultaneously --- a multi-swap move at distance 1.
        \item Exponential neighborhood size ($F_{n+1}$) explored at linear $O(n)$ cost via dynamic programming.
        \item Tridiagonal QUBO embeds cleanly on D-Wave Advantage; feasible up to $n = 500$.
        \item Empirically beats Dynasearch from $n \geq 200$, where Dynasearch's $O(n^2)$ candidate enumeration runs out of budget.
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Future work:}
    \begin{itemize}
        \item Full QPU evaluation of the Fibonacci QUBO at $n = 500$ on D-Wave Advantage.
        \item Relaxing the non-overlapping constraint --- e.g.\ minimum-gap variants.
        \item Combining Fibonacci selection with cyclic / random subset perturbation as a diversification operator.
    \end{itemize}
\end{frame}
